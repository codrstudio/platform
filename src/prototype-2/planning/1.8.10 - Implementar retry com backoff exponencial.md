### Gotchas and Pitfalls

**Thundering Herd Problem:**
- Multiple clients retry at exact same time
- Can overwhelm server after outage
- Solution: Add random jitter (±10% of delay)

**Memory Leaks:**
- Retry timers must be cleaned up
- Component unmount should cancel pending retries
- Use cleanup functions and AbortController

**Infinite Retry Loops:**
- Must have max retry limit
- Must distinguish permanent vs temporary errors
- Fatal errors (auth, validation) should not retry

**Error Classification:**
- Network timeout: Retry (transient)
- DNS failure: Retry (might recover)
- 401 Unauthorized: Don't retry (permanent)
- 429 Rate Limit: Retry with longer delay
- 500 Server Error: Retry (might be temporary)

**Timing Issues:**
- Don't retry too fast (can worsen server load)
- Don't retry too slow (poor UX)
- Balance between recovery speed and server protection

```typescript
import type {
  RetryConfig,
  RetryResult,
  RetryableError,
  RetryableErrorType,
  BackoffStrategy
} from '../types/retry';
import { isJQELError } from '../services/jqel/errors';

/**
 * Default retry configuration
 * Based on SPEC-ERR-RETRY-002:003
 */
const DEFAULT_CONFIG: Required<Omit<RetryConfig, 'signal' | 'shouldRetry' | 'onRetry'>> = {
  maxAttempts: 3,
  baseDelayMs: 1000,
  maxDelayMs: 30000,
  strategy: 'exponential',
  jitter: true,
};

/**
 * Classify error type for retry decision
 * Based on SPEC-ERR-RETRY-004:005
 */
export function classifyError(error: unknown): RetryableErrorType {
  // JQEL errors with status codes
  if (isJQELError(error)) {
    if (error.code === 429) return RetryableErrorType.RATE_LIMIT;
    if (error.code === 401 || error.code === 403) return RetryableErrorType.AUTH_ERROR;
    if (error.code >= 400 && error.code < 500) return RetryableErrorType.CLIENT_ERROR;
    if (error.code >= 500) return RetryableErrorType.SERVER_ERROR;
  }

  // Network errors
  if (error instanceof TypeError && error.message.includes('fetch')) {
    return RetryableErrorType.NETWORK_ERROR;
  }

  // Timeout errors
  if (error instanceof Error && error.name === 'AbortError') {
    return RetryableErrorType.NETWORK_ERROR;
  }

  return RetryableErrorType.UNKNOWN;
}

/**
 * Determine if error should be retried
 * Based on SPEC-ERR-RETRY-004:005
 */
export function shouldRetryError(error: unknown, attemptNumber: number): boolean {
  const errorType = classifyError(error);

  // Never retry these error types
  if (errorType === RetryableErrorType.CLIENT_ERROR ||
      errorType === RetryableErrorType.AUTH_ERROR ||
      errorType === RetryableErrorType.VALIDATION_ERROR) {
    return false;
  }

  // Always retry these (up to max attempts)
  if (errorType === RetryableErrorType.NETWORK_ERROR ||
      errorType === RetryableErrorType.SERVER_ERROR) {
    return true;
  }

  // Rate limit: retry with caution
  if (errorType === RetryableErrorType.RATE_LIMIT) {
    return attemptNumber < 2; // Fewer retries for rate limits
  }

  // Unknown errors: retry cautiously
  return attemptNumber < 1; // Only 1 retry for unknown errors
}

/**
 * Calculate backoff delay with optional jitter
 * Based on SPEC-ERR-RETRY-003
 *
 * @param attemptNumber - 0-based attempt index
 * @param config - Retry configuration
 * @returns Delay in milliseconds
 */
export function calculateBackoff(
  attemptNumber: number,
  config: Required<Omit<RetryConfig, 'signal' | 'shouldRetry' | 'onRetry'>>
): number {
  let delay: number;

  switch (config.strategy) {
    case 'exponential':
      // Exponential: 1s, 2s, 4s, 8s, 16s, 32s...
      delay = config.baseDelayMs * Math.pow(2, attemptNumber);
      break;

    case 'linear':
      // Linear: 1s, 2s, 3s, 4s, 5s...
      delay = config.baseDelayMs * (attemptNumber + 1);
      break;

    case 'constant':
      // Constant: 1s, 1s, 1s, 1s...
      delay = config.baseDelayMs;
      break;

    default:
      delay = config.baseDelayMs;
  }

  // Cap at max delay
  delay = Math.min(delay, config.maxDelayMs);

  // Add jitter (±10%) to prevent thundering herd
  if (config.jitter) {
    const jitterAmount = delay * 0.1;
    const jitter = (Math.random() * 2 - 1) * jitterAmount; // Random between -10% and +10%
    delay = Math.max(0, delay + jitter);
  }

  return Math.floor(delay);
}

/**
 * Create a retryable error with metadata
 */
export function createRetryableError(
  originalError: Error,
  attempts: number
): RetryableError {
  const error = new Error(
    `Operation failed after ${attempts} attempts: ${originalError.message}`
  ) as RetryableError;

  error.name = 'RetryableError';
  error.originalError = originalError;
  error.attempts = attempts;
  error.isRetryable = shouldRetryError(originalError, 0);

  // Maintain stack trace
  if (Error.captureStackTrace) {
    Error.captureStackTrace(error, createRetryableError);
  }

  return error;
}

/**
 * Retry an async operation with exponential backoff
 * Based on SPEC-ERR-RETRY-001:007
 *
 * @param operation - Async function to retry
 * @param config - Retry configuration
 * @returns Promise with result and retry metadata
 *
 * @example
 * // Basic usage
 * const result = await retryWithBackoff(
 *   async () => fetch('/api/data'),
 *   { maxAttempts: 3 }
 * );
 *
 * @example
 * // With custom retry logic
 * const result = await retryWithBackoff(
 *   async () => loadModule('setup'),
 *   {
 *     maxAttempts: 5,
 *     shouldRetry: (error) => error.message.includes('timeout'),
 *     onRetry: (attempt, delay) => {
 *       console.log(`Retrying in ${delay}ms (attempt ${attempt + 1})`);
 *     }
 *   }
 * );
 */
export async function retryWithBackoff<T>(
  operation: () => Promise<T>,
  userConfig: RetryConfig = {}
): Promise<RetryResult<T>> {
  const config = { ...DEFAULT_CONFIG, ...userConfig };
  const startTime = Date.now();

  let lastError: Error | null = null;
  let attemptNumber = 0;

  while (attemptNumber <= config.maxAttempts) {
    // Check if operation was aborted
    if (config.signal?.aborted) {
      throw new Error('Retry operation aborted');
    }

    try {
      // Execute operation
      const value = await operation();

      // Success!
      return {
        value,
        attempts: attemptNumber + 1,
        totalTimeMs: Date.now() - startTime,
        wasRetried: attemptNumber > 0
      };

    } catch (error) {
      lastError = error instanceof Error ? error : new Error(String(error));

      // Check if we should retry
      const customShouldRetry = userConfig.shouldRetry?.(lastError, attemptNumber);
      const defaultShouldRetry = shouldRetryError(lastError, attemptNumber);
      const shouldRetry = customShouldRetry ?? defaultShouldRetry;

      // If this is the last attempt or error is not retryable, throw
      if (attemptNumber >= config.maxAttempts || !shouldRetry) {
        throw createRetryableError(lastError, attemptNumber + 1);
      }

      // Calculate delay for next retry
      const delay = calculateBackoff(attemptNumber, config);

      // Log retry in development
      if (import.meta.env.DEV) {
        console.log(
          `[Retry] Attempt ${attemptNumber + 1}/${config.maxAttempts} failed. ` +
          `Retrying in ${delay}ms. Error: ${lastError.message}`
        );
      }

      // Invoke retry callback
      userConfig.onRetry?.(attemptNumber, delay);

      // Wait before next attempt
      await new Promise<void>((resolve, reject) => {
        const timeoutId = setTimeout(resolve, delay);

        // Support cancellation via AbortSignal
        if (config.signal) {
          config.signal.addEventListener('abort', () => {
            clearTimeout(timeoutId);
            reject(new Error('Retry operation aborted'));
          }, { once: true });
        }
      });

      attemptNumber++;
    }
  }

  // Should never reach here, but TypeScript needs it
  throw createRetryableError(
    lastError || new Error('Unknown error'),
    attemptNumber
  );
}

/**
 * Retry with exponential backoff (simpler API)
 * Alias for retryWithBackoff with just max attempts
 */
export async function retry<T>(
  operation: () => Promise<T>,
  maxAttempts: number = 3
): Promise<T> {
  const result = await retryWithBackoff(operation, { maxAttempts });
  return result.value;
}
```

```typescript
import { useCallback, useRef, useEffect, useState } from 'react';
import { retryWithBackoff } from '../utils/retry';
import type { RetryConfig, RetryResult } from '../types/retry';

export interface UseRetryOptions<T> extends RetryConfig {
  /** Callback on successful completion (including retries) */
  onSuccess?: (result: RetryResult<T>) => void;
  /** Callback on final error (after all retries exhausted) */
  onError?: (error: Error) => void;
}

export interface UseRetryReturn<T> {
  /** Execute operation with retry */
  execute: (operation: () => Promise<T>) => Promise<RetryResult<T>>;
  /** Current retry state */
  isRetrying: boolean;
  /** Current attempt number (0 if not retrying) */
  currentAttempt: number;
  /** Cancel ongoing retry operation */
  cancel: () => void;
}

/**
 * React hook for retry with automatic cleanup
 * Based on SPEC-ERR-RETRY-001:007
 *
 * @example
 * const { execute, isRetrying, currentAttempt, cancel } = useRetry<Portal>({
 *   maxAttempts: 5,
 *   onSuccess: (result) => {
 *     console.log(`Loaded after ${result.attempts} attempts`);
 *   },
 *   onError: (error) => {
 *     toast({ title: 'Failed to load', variant: 'error' });
 *   }
 * });
 *
 * // Execute with retry
 * await execute(() => fetchPortal('main'));
 *
 * // Cancel if component unmounts
 * useEffect(() => {
 *   return () => cancel();
 * }, []);
 */
export function useRetry<T = unknown>(
  options: UseRetryOptions<T> = {}
): UseRetryReturn<T> {
  const [isRetrying, setIsRetrying] = useState(false);
  const [currentAttempt, setCurrentAttempt] = useState(0);
  const abortControllerRef = useRef<AbortController | null>(null);

  /**
   * Cancel ongoing retry operation
   */
  const cancel = useCallback(() => {
    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
      abortControllerRef.current = null;
    }
    setIsRetrying(false);
    setCurrentAttempt(0);
  }, []);

  /**
   * Execute operation with retry
   */
  const execute = useCallback(async (
    operation: () => Promise<T>
  ): Promise<RetryResult<T>> => {
    // Cancel any previous operation
    cancel();

    // Create new AbortController for this operation
    const abortController = new AbortController();
    abortControllerRef.current = abortController;

    setIsRetrying(true);
    setCurrentAttempt(0);

    try {
      const result = await retryWithBackoff(operation, {
        ...options,
        signal: abortController.signal,
        onRetry: (attemptNumber, delay) => {
          setCurrentAttempt(attemptNumber + 1);
          options.onRetry?.(attemptNumber, delay);
        }
      });

      setIsRetrying(false);
      setCurrentAttempt(0);
      options.onSuccess?.(result);

      return result;

    } catch (error) {
      setIsRetrying(false);
      setCurrentAttempt(0);
      options.onError?.(error as Error);
      throw error;

    } finally {
      abortControllerRef.current = null;
    }
  }, [options, cancel]);

  // Cleanup on unmount
  useEffect(() => {
    return () => {
      cancel();
    };
  }, [cancel]);

  return {
    execute,
    isRetrying,
    currentAttempt,
    cancel
  };
}
```

```typescript
import type {
  RetryConfig,
  RetryResult,
  BackoffStrategy
} from '../types/retry.types';

const DEFAULT_CONFIG = {
  maxAttempts: 3,
  baseDelayMs: 1000,
  maxDelayMs: 30000,
  strategy: 'exponential' as BackoffStrategy,
  jitter: true,
};

/**
 * Calculate exponential backoff delay
 */
function calculateBackoff(
  attemptNumber: number,
  config: typeof DEFAULT_CONFIG
): number {
  let delay: number;

  switch (config.strategy) {
    case 'exponential':
      delay = config.baseDelayMs * Math.pow(2, attemptNumber);
      break;
    case 'linear':
      delay = config.baseDelayMs * (attemptNumber + 1);
      break;
    case 'constant':
      delay = config.baseDelayMs;
      break;
  }

  delay = Math.min(delay, config.maxDelayMs);

  if (config.jitter) {
    const jitterAmount = delay * 0.1;
    const jitter = (Math.random() * 2 - 1) * jitterAmount;
    delay = Math.max(0, delay + jitter);
  }

  return Math.floor(delay);
}

/**
 * Retry async operation with exponential backoff
 * Simplified backend version
 */
export async function retryWithBackoff<T>(
  operation: () => Promise<T>,
  userConfig: Partial<RetryConfig> = {}
): Promise<RetryResult<T>> {
  const config = { ...DEFAULT_CONFIG, ...userConfig };
  const startTime = Date.now();

  let lastError: Error | null = null;
  let attemptNumber = 0;

  while (attemptNumber <= config.maxAttempts) {
    try {
      const value = await operation();

      return {
        value,
        attempts: attemptNumber + 1,
        totalTimeMs: Date.now() - startTime,
        wasRetried: attemptNumber > 0
      };

    } catch (error) {
      lastError = error instanceof Error ? error : new Error(String(error));

      if (attemptNumber >= config.maxAttempts) {
        throw lastError;
      }

      const delay = calculateBackoff(attemptNumber, config);

      console.log(
        `[Retry] Attempt ${attemptNumber + 1}/${config.maxAttempts} failed. ` +
        `Retrying in ${delay}ms. Error: ${lastError.message}`
      );

      await new Promise(resolve => setTimeout(resolve, delay));
      attemptNumber++;
    }
  }

  throw lastError || new Error('Unknown error');
}
```
