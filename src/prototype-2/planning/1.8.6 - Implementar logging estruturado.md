### Gotchas and Pitfalls

**⚠️ Context Hook Usage Outside Provider**
- Calling `useAuth()` outside AuthProvider throws error
- Must wrap context access in try-catch or check if context exists
- Alternative: Pass context explicitly rather than fetching globally

**⚠️ Circular References in Context**
- Logging entire React objects can cause circular reference errors
- Must extract only serializable primitive values
- Use `JSON.stringify()` with try-catch to detect non-serializable data

**⚠️ Performance Impact of Context Enrichment**
- Avoid expensive operations in log context generation
- Don't fetch data from APIs just for logging context
- Consider lazy evaluation for optional context fields

**⚠️ Regex Performance for Sanitization**
- Complex regex patterns on large objects can be slow
- Apply sanitization only to string values, not entire object
- Use simple string matching for common patterns first

**⚠️ TypeScript Strict Null Checks**
- Context fields are optional (`userId?: string`)
- Use optional chaining (`user?.id`) to avoid null errors
- Provide fallback values for critical fields

```typescript
/**
 * Sanitizer - Remove/mask sensitive data from log context
 * Based on SPEC-EH-LO-005
 */

import type { LogContext, SanitizeOptions } from '../../types/errorBoundary';

/**
 * Default sensitive field names (case-insensitive)
 */
const DEFAULT_SENSITIVE_FIELDS = [
  'password',
  'token',
  'secret',
  'apikey',
  'api_key',
  'authorization',
  'auth',
  'jwt',
  'refreshtoken',
  'refresh_token',
  'accesstoken',
  'access_token',
  'sessiontoken',
  'session_token',
  'privatekey',
  'private_key',
  'credential',
  'credentials'
];

/**
 * Check if field name indicates sensitive data
 */
export function isSensitiveField(fieldName: string, customFields?: string[]): boolean {
  const lowerName = fieldName.toLowerCase();
  const allFields = [...DEFAULT_SENSITIVE_FIELDS, ...(customFields || [])];
  return allFields.some(sensitive => lowerName.includes(sensitive));
}

/**
 * Mask email address (e.g., user@example.com → u***@e***.com)
 */
export function maskEmail(email: string): string {
  const parts = email.split('@');
  if (parts.length !== 2) return '[REDACTED]';

  const [user, domain] = parts;
  const maskedUser = user.length > 1 ? user[0] + '***' : '***';
  const domainParts = domain.split('.');
  const maskedDomain = domainParts.length > 1
    ? domainParts[0][0] + '***.' + domainParts[domainParts.length - 1]
    : '***';

  return `${maskedUser}@${maskedDomain}`;
}

/**
 * Mask a sensitive string value
 */
export function maskValue(value: any, fieldName?: string): string {
  if (typeof value !== 'string') return '[REDACTED]';

  // Special handling for email-like patterns
  if (fieldName?.toLowerCase().includes('email') || value.includes('@')) {
    return maskEmail(value);
  }

  // Generic masking: Show first 2 and last 2 characters
  if (value.length <= 4) return '***';
  return `${value.slice(0, 2)}***${value.slice(-2)}`;
}

/**
 * Sanitize context object by removing/masking sensitive fields
 * Based on SPEC-EH-LO-005
 *
 * @param context - Context object to sanitize
 * @param options - Sanitization options
 * @returns Sanitized context object
 */
export function sanitizeContext(
  context: Record<string, any>,
  options?: SanitizeOptions
): Record<string, any> {
  const {
    redactFields = [],
    maskFields = [],
    maskEmails = true
  } = options || {};

  const sanitized: Record<string, any> = {};

  for (const [key, value] of Object.entries(context)) {
    // Skip null/undefined
    if (value === null || value === undefined) {
      sanitized[key] = value;
      continue;
    }

    // Check if field should be completely redacted
    if (isSensitiveField(key, redactFields)) {
      sanitized[key] = '[REDACTED]';
      continue;
    }

    // Check if field should be masked
    if (maskFields.includes(key.toLowerCase())) {
      sanitized[key] = maskValue(value, key);
      continue;
    }

    // Handle nested objects (recursive sanitization)
    if (typeof value === 'object' && !Array.isArray(value)) {
      try {
        sanitized[key] = sanitizeContext(value, options);
      } catch (error) {
        // Handle circular references
        sanitized[key] = '[CIRCULAR]';
      }
      continue;
    }

    // Handle arrays
    if (Array.isArray(value)) {
      sanitized[key] = value.map(item =>
        typeof item === 'object' ? sanitizeContext(item, options) : item
      );
      continue;
    }

    // Mask emails in string values
    if (maskEmails && typeof value === 'string' && value.includes('@')) {
      sanitized[key] = maskEmail(value);
      continue;
    }

    // Keep non-sensitive values as-is
    sanitized[key] = value;
  }

  return sanitized;
}
```

/**
 * Log Context Utilities - Auto-enrich log entries with contextual data
 * Based on SPEC-EH-LO-015:018
 */

import type { LogContext } from '../../types/errorBoundary';

/**
 * Get current portal ID from URL
 * Based on routing pattern: "/" → "main", "/:portalId/*" → portalId
 */
export function getCurrentPortalId(): string {
  const path = window.location.pathname;

  // Root path = main portal
  if (path === '/') return 'main';

  // Extract first segment after /
  const segments = path.split('/').filter(Boolean);
  return segments[0] || 'main';
}

/**
 * Get current module ID from URL or route context
 * Note: This is best-effort, may not always be accurate
 */
export function getCurrentModuleId(): string | undefined {
  // Try to extract from URL path
  const path = window.location.pathname;
  const segments = path.split('/').filter(Boolean);

  // Pattern: /portalId/moduleId/...
  if (segments.length >= 2) {
    return segments[1];
  }

  return undefined;
}

/**
 * Get user context from AuthProvider (if available)
 * Note: This can only be called from within React components/hooks
 * For use outside React, pass user context explicitly
 */
export function getUserContext(): Pick<LogContext, 'userId' | 'username' | 'realm'> | undefined {
  try {
    // Try to read from localStorage (where AuthProvider stores user info)
    const userStr = localStorage.getItem('user');
    if (!userStr) return undefined;

    const user = JSON.parse(userStr);
    return {
      userId: user.id,
      username: user.username,
      realm: user.realm
    };
  } catch (error) {
    // Failed to get user context (not authenticated or storage error)
    return undefined;
  }
}

/**
 * Get automatic log context from current environment
 * Based on SPEC-EH-LO-015:018
 *
 * Gathers context from:
 * - Current user (if authenticated)
 * - Current portal (from URL)
 * - Current module (from URL)
 * - Browser info (URL, user agent)
 *
 * @returns LogContext with automatic fields populated
 */
export function getLogContext(): LogContext {
  const userContext = getUserContext();

  return {
    // User context
    ...userContext,

    // Portal/Module context
    portalId: getCurrentPortalId(),
    moduleId: getCurrentModuleId(),

    // Technical context
    url: window.location.href,
    userAgent: navigator.userAgent,

    // Session ID (can be generated or from session storage)
    sessionId: sessionStorage.getItem('sessionId') || undefined
  };
}

/**
 * Enrich provided context with automatic context
 * Merges user-provided context with automatically gathered context
 *
 * @param userContext - User-provided context
 * @param autoContext - Auto-gathered context (optional, will call getLogContext if not provided)
 * @returns Merged context
 */
export function enrichContext(
  userContext?: Record<string, any>,
  autoContext?: LogContext
): Record<string, any> {
  const auto = autoContext || getLogContext();

  // User context takes precedence over auto context
  return {
    ...auto,
    ...userContext
  };
}

/**
 * Log error with structured data
 * Based on SPEC-DA-ERR-008 and SPEC-EH-LO-015:018
 *
 * @param error - Error object to log
 * @param options - Logging options
 *
 * @example
 * logError(new Error('Query failed'), {
 *   category: 'jqel-query',
 *   level: 'ERROR',
 *   context: { queryKey: 'portal:main' },
 *   autoEnrich: true,  // Auto-add user/portal/module context
 *   autoSanitize: true // Auto-remove sensitive data
 * });
 */
export function logError(error: Error, options: LogErrorOptions): void {
  const {
    category,
    level = 'ERROR',
    context = {},
    autoEnrich = true,
    autoSanitize = true
  } = options;

  // Enrich context with automatic data
  let enrichedContext = autoEnrich ? enrichContext(context) : context;

  // Sanitize sensitive data
  if (autoSanitize) {
    enrichedContext = sanitizeContext(enrichedContext);
  }

  const logEntry: ErrorLogEntry = {
    timestamp: new Date().toISOString(),
    level,
    category,
    message: error.message,
    error: {
      name: error.name,
      message: error.message,
      ...(import.meta.env.DEV && { stack: error.stack })
    },
    context: enrichedContext
  };

  // Environment-aware output with appropriate console method
  if (import.meta.env.DEV) {
    switch (level) {
      case 'ERROR':
        console.error(`[${level}] ${category}:`, logEntry);
        break;
      case 'WARN':
        console.warn(`[${level}] ${category}:`, logEntry);
        break;
      case 'INFO':
        console.info(`[${level}] ${category}:`, logEntry);
        break;
      case 'DEBUG':
        console.debug(`[${level}] ${category}:`, logEntry);
        break;
    }
  } else if (level === 'ERROR' || level === 'WARN') {
    if (level === 'ERROR') {
      console.error(`[${level}] ${category}:`, logEntry);
    } else {
      console.warn(`[${level}] ${category}:`, logEntry);
    }
  }

  // TODO: Send to monitoring service in production (SPEC-EH-LO-010)
