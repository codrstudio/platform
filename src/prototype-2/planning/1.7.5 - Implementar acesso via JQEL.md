```typescript
export function useJQELQuery<T = any>(
  query: JQELQuery,
  options?: UseJQELQueryOptions<T>
): UseJQELQueryResult<T> {
  const queryKey = options?.queryKey || generateQueryKey(query);

  const result = useQuery<T[], JQELError>({
    queryKey,
    queryFn: () => jqelQuery<T>(query),
    ...DEFAULT_OPTIONS,
    ...options,
  });

  return {
    data: result.data?.[0] ?? null, // Extract first item by default
    isLoading: result.isLoading,
    error: result.error ?? null,
    refetch: result.refetch,
    // ... other fields
  };
}
```

### Gotchas and Pitfalls

- **⚠️ Array vs single object**: `useJQELQuery` extracts first item by default - list queries need `allData: true`
- **⚠️ Query key consistency**: Must use factory functions (not inline arrays) for cache invalidation to work
- **⚠️ WHERE clause syntax**: Must use operator objects: `{ portalId: { $eq: 'main' } }` not `{ portalId: 'main' }`
- **⚠️ Backend schema routing**: Only works with `schema: 'backend'` - other schemas route to n8n
- **⚠️ Mutation order**: UPDATE requires WHERE clause, INSERT does not - validation must check this
- **⚠️ Optimistic updates**: Snapshot must capture entire entity list, not just modified item

### Existing Patterns to Follow

**Query options pattern** - See `frontend/src/services/jqel/hooks/useJQELQuery.ts` (lines 63-69)
- Default `staleTime: 5 minutes` for queries
- Default `gcTime: 10 minutes` for cache retention
- Disable `refetchOnWindowFocus` (configuration changes are infrequent)

**Invalidation scope pattern** - See `frontend/src/services/jqel/invalidation.ts` (lines 48-53)
- INSERT: Invalidate entity (entire list)
- UPDATE: Invalidate specific (if ID known) + entity
- DELETE: Invalidate entity (entire list)

**Error typing pattern** - See `frontend/src/services/jqel/errors.ts`
- Use `JQELError` interface with `code`, `message`, `field?`

## Technical Specification

### Architecture

```
Frontend Query Flow:
Component
    ↓ (calls hook)
usePortals()
    ↓ (generates query key)
['jqel', 'backend', 'portals']
    ↓ (calls TanStack Query)
useQuery({ queryKey, queryFn })
    ↓ (executes JQEL)
jqelQuery({ schema: 'backend', select: 'portal' })
    ↓ (POST /api/jqel)
Backend JQEL Endpoint
    ↓ (validates query)
JQEL Validation Middleware
    ↓ (routes by schema)
JqelRouter.route()
    ↓ (schema === 'backend')
BackendProcessor.process()
    ↓ (reads file)
fs.readFile('config/portals.json')
    ↓ (applies filters)
WHERE, ORDER BY, LIMIT
    ↓ (returns JResult)
{ code: 200, data: [...] }
```

### Data Flow

```
Frontend Mutation Flow:
Component calls useCreatePortal()
    ↓
Mutation executes
    ↓ (optimistic update)
Snapshot current cache
    ↓
Update cache immediately (optimistic)
    ↓
POST /api/jqel (mutation)
    ↓ (success)
Invalidate queries
    ↓
Refetch updated data
    ↓ (error)
Rollback cache to snapshot
    ↓
Show error to user
```

### Error Handling Strategy

**Error Types:**

1. **Invalid entity name**
   - Handling: Middleware validation returns 400
   - User feedback: "Invalid entity: [name]"
   - Example: `{ schema: 'backend', select: 'invalid' }` → 400 error

2. **Missing WHERE clause (UPDATE/DELETE)**
   - Handling: BackendProcessor returns 400
   - User feedback: "WHERE clause is required for update/delete"
   - Example: `{ schema: 'backend', mutate: 'portal', action: 'update', values: {...} }` → 400 error

3. **File read error**
   - Handling: BackendProcessor logs error, returns 500
   - User feedback: "Failed to load configuration"
   - Recovery: Retry query (automatic via TanStack Query)

4. **File write error (mutation)**
   - Handling: BackendProcessor logs error, returns 500
   - User feedback: "Failed to save configuration"
   - Recovery: Rollback optimistic update, show error message

5. **Network error**
   - Handling: JQEL client throws JQELError
   - User feedback: "Network error, retrying..."
   - Recovery: Automatic retry with exponential backoff (up to 3 attempts)

**Error Display Pattern:**

```typescript
// Component usage pattern
const { data: portal, error, isLoading } = usePortal('main');

if (isLoading) return <Spinner />;
if (error) return <ErrorMessage error={error} />;
if (!portal) return <NotFound />;

return <PortalDetails portal={portal} />;
```

**Optimistic Update Rollback:**

```typescript
// Pattern from: frontend/src/hooks/jqel/optimisticHelpers.ts
const snapshot = captureSnapshot(queryClient, queryKey);

try {
  // Apply optimistic update
  queryClient.setQueryData(queryKey, optimisticData);

  // Execute mutation
  await mutationFn();
} catch (error) {
  // Rollback on error
  restoreSnapshot(queryClient, queryKey, snapshot);
  throw error;
}
```

**`frontend/src/hooks/jqel/usePortalQueries.ts`** - Portal query hooks
```typescript
import { useJQELQuery } from '../../services/jqel/hooks/useJQELQuery';
import { queryKeys } from '../../services/jqel/queryKeys';
import type { Portal } from '../../types/config';

/**
 * Fetch all portals from backend schema
 *
 * @example
 * const { data: portals, isLoading } = usePortals();
 */
export function usePortals() {
  return useJQELQuery<Portal[]>({
    schema: 'backend',
    select: 'portal',
  }, {
    queryKey: queryKeys.backend.portals(),
    select: (data) => data, // Return all items (not just first)
  });
}

/**
 * Fetch single portal by ID
 *
 * @param portalId - Portal ID to fetch
 * @example
 * const { data: portal, isLoading } = usePortal('main');
 */
export function usePortal(portalId: string) {
  return useJQELQuery<Portal>({
    schema: 'backend',
    select: 'portal',
    where: { portalId: { $eq: portalId } },
  }, {
    queryKey: queryKeys.backend.portal(portalId),
    enabled: !!portalId, // Don't fetch if ID is empty
  });
}

/**
 * Fetch only active portals
 *
 * @example
 * const { data: activePortals } = useActivePortals();
 */
export function useActivePortals() {
  return useJQELQuery<Portal[]>({
    schema: 'backend',
    select: 'portal',
    where: { active: { $eq: true } },
  }, {
    queryKey: [...queryKeys.backend.portals(), 'active'],
    select: (data) => data, // Return all items
  });
}
```

**`frontend/src/hooks/jqel/usePortalMutations.ts`** - Portal mutation hooks
```typescript
import { useQueryClient } from '@tanstack/react-query';
import { useOptimisticMutation } from './useOptimisticMutation';
import { jqelMutate } from '../../services/jqel/mutations';
import { queryKeys } from '../../services/jqel/queryKeys';
import type { Portal } from '../../types/config';

/**
 * Create new portal
 *
 * @example
 * const createPortal = useCreatePortal();
 * createPortal.mutate({ portalId: 'shop', name: 'Shop Portal', ... });
 */
export function useCreatePortal() {
  const queryClient = useQueryClient();

  return useOptimisticMutation<Portal>({
    mutationFn: async (portal: Portal) => {
      return jqelMutate({
        schema: 'backend',
        mutate: 'portal',
        action: 'insert',
        values: portal,
      });
    },
    onSuccess: () => {
      queryClient.invalidateQueries({
        queryKey: queryKeys.backend.portals(),
      });
    },
  });
}

/**
 * Update existing portal
 *
 * @example
 * const updatePortal = useUpdatePortal();
 * updatePortal.mutate({ portalId: 'main', values: { name: 'New Name' } });
 */
export function useUpdatePortal() {
  const queryClient = useQueryClient();

  return useOptimisticMutation<{ portalId: string; values: Partial<Portal> }>({
    mutationFn: async ({ portalId, values }) => {
      return jqelMutate({
        schema: 'backend',
        mutate: 'portal',
        action: 'update',
        where: { portalId: { $eq: portalId } },
        values,
      });
    },
    onSuccess: (_, { portalId }) => {
      // Invalidate specific portal + list
      queryClient.invalidateQueries({
        queryKey: queryKeys.backend.portal(portalId),
      });
      queryClient.invalidateQueries({
        queryKey: queryKeys.backend.portals(),
      });
    },
  });
}

/**
 * Delete portal
 *
 * @example
 * const deletePortal = useDeletePortal();
 * deletePortal.mutate('shop');
 */
export function useDeletePortal() {
  const queryClient = useQueryClient();

  return useOptimisticMutation<string>({
    mutationFn: async (portalId: string) => {
      return jqelMutate({
        schema: 'backend',
        mutate: 'portal',
        action: 'delete',
        where: { portalId: { $eq: portalId } },
      });
    },
    onSuccess: () => {
      queryClient.invalidateQueries({
        queryKey: queryKeys.backend.portals(),
      });
    },
  });
}
```
