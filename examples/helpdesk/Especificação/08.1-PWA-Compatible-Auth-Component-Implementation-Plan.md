- [ ] Criar `src/core/auth-sw.ts`
  ```typescript
  /// <reference lib="webworker" />

  import { storage } from './storage'

  declare const self: ServiceWorkerGlobalScope

  // Intercepta requests /api/*
  self.addEventListener('fetch', (event: FetchEvent) => {
    const url = new URL(event.request.url)

    if (!url.pathname.startsWith('/api/')) {
      return
    }

    event.respondWith(
      (async () => {
        const authData = await storage.get()

        if (!authData?.access_token) {
          return fetch(event.request)
        }

        const headers = new Headers(event.request.headers)
        headers.set('Authorization', `Bearer ${authData.access_token}`)

        const authRequest = new Request(event.request, { headers })
        return fetch(authRequest)
      })()
    )
  })
  ```

### Tarefa 5.1: Criar AuthProvider

- [ ] Criar `src/context/AuthProvider.tsx`
  ```typescript
  import { createContext, useContext, useState, useEffect, ReactNode } from 'react'
  import { storage } from '../core/storage'
  import { sync } from '../core/sync'
  import { registerServiceWorker } from '../core/register-sw'
  import type { UserPayload } from '../types'

  interface AuthContextType {
    user: UserPayload | null
    loading: boolean
    login: (email: string, password: string) => Promise<{
      success: boolean
      message?: string
    }>
    logout: () => Promise<void>
  }

  const AuthContext = createContext<AuthContextType | undefined>(undefined)

  export function AuthProvider({ children }: { children: ReactNode }) {
    const [user, setUser] = useState<UserPayload | null>(null)
    const [loading, setLoading] = useState(true)

    useEffect(() => {
      initialize()
    }, [])

    async function initialize() {
      // 1. Carrega auth do IndexedDB
      const authData = await storage.get()
      setUser(authData?.payload || null)
      setLoading(false)

      // 2. Registra Service Worker
      await registerServiceWorker()

      // 3. Escuta eventos de outros apps
      sync.listen((event) => {
        if (event.type === 'LOGIN') {
          setUser(event.payload)
        } else if (event.type === 'LOGOUT') {
          setUser(null)
        }
      })
    }

    async function login(email: string, password: string) {
      try {
        const res = await fetch('/webhook/coletivos/api/1/autenticar', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ username: email, password })
        })

        const json = await res.json()

        if (json.code !== 200 || !json.data) {
          return {
            success: false,
            message: json.message || 'Erro ao fazer login'
          }
        }

        const { access_token, payload, expires_in } = json.data

        await storage.save({
          access_token,
          payload,
          expires_at: Date.now() + expires_in * 1000
        })

        setUser(payload)
        sync.broadcast({ type: 'LOGIN', payload, token: access_token })

        // Sincroniza tema
        const tema = payload.preferencias.tema === 'claro' ? 'light' : 'dark'
        localStorage.setItem('pref-theme', tema)

        return { success: true }
      } catch (error) {
        return {
          success: false,
          message: 'Erro de conexão'
        }
      }
    }

    async function logout() {
      await storage.clear()
      setUser(null)
      sync.broadcast({ type: 'LOGOUT' })
    }

    return (
      <AuthContext.Provider value={{ user, loading, login, logout }}>
        {children}
      </AuthContext.Provider>
    )
  }

  export function useAuth() {
    const context = useContext(AuthContext)
    if (!context) {
      throw new Error('useAuth must be used within AuthProvider')
    }
    return context
  }
  ```

### Tarefa 6.1: Criar ThemeProvider

- [ ] Criar `src/context/ThemeProvider.tsx`
  ```typescript
  import { createContext, useContext, useState, useEffect, ReactNode } from 'react'
  import { sync } from '../core/sync'

  const THEME_KEY = 'pref-theme'

  type Theme = 'light' | 'dark'

  interface ThemeContextType {
    theme: Theme
    setTheme: (theme: Theme) => void
    toggleTheme: () => void
  }

  const ThemeContext = createContext<ThemeContextType | undefined>(undefined)

  export function ThemeProvider({ children }: { children: ReactNode }) {
    const [theme, setThemeState] = useState<Theme>(() => {
      const saved = localStorage.getItem(THEME_KEY) as Theme | null
      return saved || 'light'
    })

    useEffect(() => {
      // Aplica no DOM
      document.documentElement.setAttribute('data-theme', theme)
      document.documentElement.classList.toggle('dark', theme === 'dark')

      // Salva
      localStorage.setItem(THEME_KEY, theme)

      // Notifica outros apps
      sync.broadcast({ type: 'THEME_CHANGE', theme })
    }, [theme])

    useEffect(() => {
      // Escuta mudanças de outros apps
      sync.listen((event) => {
        if (event.type === 'THEME_CHANGE') {
          setThemeState(event.theme)
        }
      })
    }, [])

    const setTheme = (newTheme: Theme) => {
      setThemeState(newTheme)
    }

    const toggleTheme = () => {
      setTheme(theme === 'light' ? 'dark' : 'light')
    }

    return (
      <ThemeContext.Provider value={{ theme, setTheme, toggleTheme }}>
        {children}
      </ThemeContext.Provider>
    )
  }

  export function useTheme() {
    const context = useContext(ThemeContext)
    if (!context) {
      throw new Error('useTheme must be used within ThemeProvider')
    }
    return context
  }
  ```

  ```typescript
          <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor">
            <path d="M12 3a6 6 0 0 0 9 9 9 9 0 1 1-9-9Z" />
          </svg>
        ) : (
          // Sol (ativa light)
          <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor">
            <circle cx="12" cy="12" r="4" />
            <path d="M12 2v2" />
            <path d="M12 20v2" />
            <path d="m4.93 4.93 1.41 1.41" />
            <path d="m17.66 17.66 1.41 1.41" />
            <path d="M2 12h2" />
            <path d="M20 12h2" />
            <path d="m6.34 17.66-1.41 1.41" />
            <path d="m19.07 4.93-1.41 1.41" />
          </svg>
        )}
      </button>
    )
  }
  ```

### Tarefa 7.4: Criar ProtectedRoute

- [ ] Criar `src/components/ProtectedRoute.tsx`
  ```typescript
  import { ReactNode } from 'react'
  import { useAuth } from '../context/AuthProvider'

  interface ProtectedRouteProps {
    children: ReactNode
    fallback?: ReactNode
    redirectToLogin?: () => void
  }

  export function ProtectedRoute({
    children,
    fallback,
    redirectToLogin
  }: ProtectedRouteProps) {
    const { user, loading } = useAuth()

    if (loading) {
      return (
        fallback || (
          <div style={{
            display: 'flex',
            minHeight: '100vh',
            alignItems: 'center',
            justifyContent: 'center'
          }}>
            Carregando...
          </div>
        )
      )
    }

    if (!user) {
      if (redirectToLogin) {
        redirectToLogin()
        return null
      }

      return (
        fallback || (
          <div style={{
            display: 'flex',
            minHeight: '100vh',
            alignItems: 'center',
            justifyContent: 'center'
          }}>
            Não autorizado
          </div>
        )
      )
    }

    return <>{children}</>
  }
  ```

```typescript
// src/helpdesk/src/components/UserProfile.tsx
import { useAuth, useTheme } from '@coletivos/auth-client'

export function UserProfile() {
  const { user, logout } = useAuth()
  const { theme, toggleTheme } = useTheme()

  return (
    <div>
      <h2>Olá, {user.nome_exibicao}</h2>

      <button onClick={toggleTheme}>

      {user.is_admin && <p>⭐ Administrador</p>}

        {user.papeis.map(papel => (
          <li key={papel.codigo}>

          </li>
        ))}
      </ul>
    </div>
  )
}
```

### Service Worker Automático

O Service Worker é registrado automaticamente pelo `AuthProvider`. Todas as requisições para `/api/*` receberão o header `Authorization: Bearer {token}` automaticamente.

```typescript
// Suas chamadas API normais

axios.get('/api/usuarios')  // ← Token injetado automaticamente pelo SW
```
